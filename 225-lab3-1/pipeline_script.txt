\f0\fs24 \cf0 pipeline \{\
    agent any \
\
    environment \{\
        // Define your DockerHub credentialsId, repo name, and tag name\
        DOCKER_CREDENTIALS_ID = 'roseaw-dockerhub'\
        DOCKER_IMAGE = 'meltoneb/lab3-1'\
        GIT_URL=['https://github.com/meltoneb-glitch/CIT-225-Repository-Melton']\
        IMAGE_TAG = 'latest'\
        // The following credentials do not need to be changed!!!\
        KUBECONFIG = credentials('kube-01')\
    \}\
\
    stages \{\
\
        stage('Checkout') \{\
            steps \{\
                checkout([$class: 'GitSCM', branches: [[name: '*/main']],\
                          userRemoteConfigs: [url:GIT_URL])\
            \}\
        \}\
\
        stage('Build Docker Image') \{\
            steps \{\
                // Build Docker Image\
                script \{\
                    docker.build("$\{DOCKER_IMAGE\}:$\{IMAGE_TAG\}")\
                \}\
            \}\
        \}\
   \
        stage('Push Docker Image') \{\
            steps \{\
                script \{\
                    docker.withRegistry('https://index.docker.io/v1/', "$\{DOCKER_CREDENTIALS_ID\}") \{\
                        docker.image("$\{DOCKER_IMAGE\}:$\{IMAGE_TAG\}").push()\
                    \}\
                \}\
            \}\
        \}\
\
        stage('Deploy to Kubernetes') \{\
            steps \{\
                script \{\
                    def kubeConfig = readFile(KUBECONFIG)\
                    sh "kubectl --kubeconfig=\\$KUBECONFIG apply -f deployment.yaml"\
                \}\
            \}\
        \}\
    \}\
\}\
